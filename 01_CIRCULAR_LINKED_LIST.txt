CIRCULAR LINKED LIST
====================

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;
};

struct Node *last = NULL;

void insertBeginning(int x) {
    struct Node *newNode = malloc(sizeof(struct Node));
    if (!newNode) return;
    newNode->data = x;

    if (last == NULL) {
        last = newNode;
        newNode->next = newNode;
    } else {
        newNode->next = last->next;
        last->next = newNode;
    }
}

void insertEnd(int x) {
    insertBeginning(x);
    last = last->next;
}

void deleteNode(int x) {
    if (last == NULL) {
        printf("List is empty\n");
        return;
    }

    struct Node *prev = last, *cur = last->next;
    do {
        if (cur->data == x) {
            if (cur == prev) {             /* only one node */
                free(cur);
                last = NULL;
            } else {
                prev->next = cur->next;
                if (cur == last) last = prev;
                free(cur);
            }
            return;
        }
        prev = cur;
        cur = cur->next;
    } while (cur != last->next);

    printf("Element not found\n");
}

void display(void) {
    if (last == NULL) {
        printf("List is empty\n");
        return;
    }
    struct Node *temp = last->next;
    printf("Circular Linked List: ");
    do {
        printf("%d -> ", temp->data);
        temp = temp->next;
    } while (temp != last->next);
    printf("(back to first)\n");
}

int main(void) {
    int ch, x;
    while (1) {
        printf("\n1. Insert at beginning\n2. Insert at end\n3. Delete\n4. Display\n5. Exit\nChoice: ");
        if (scanf("%d", &ch) != 1) return 0;
        switch (ch) {
            case 1: printf("Enter data: "); scanf("%d", &x); insertBeginning(x); break;
            case 2: printf("Enter data: "); scanf("%d", &x); insertEnd(x); break;
            case 3: printf("Enter data to delete: "); scanf("%d", &x); deleteNode(x); break;
            case 4: display(); break;
            case 5: return 0;
            default: printf("Invalid choice\n");
        }
    }
}
