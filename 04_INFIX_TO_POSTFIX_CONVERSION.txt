INFIX TO POSTFIX CONVERSION
===========================

#include <stdio.h>
#include <ctype.h>
#include <string.h>

#define MAX 100
char stack[MAX];
int top = -1;

void push(char c) { if (top < MAX - 1) stack[++top] = c; }
char pop(void) { return top >= 0 ? stack[top--] : '\0'; }
char peek(void) { return top >= 0 ? stack[top] : '\0'; }

int precedence(char c) {
    if (c == '^') return 3;
    if (c == '*' || c == '/') return 2;
    if (c == '+' || c == '-') return 1;
    return 0;
}

int main(void) {
    char infix[MAX], postfix[MAX];
    int i, k = 0;
    printf("Enter infix expression (no spaces): ");
    scanf("%99s", infix);

    for (i = 0; infix[i] != '\0'; i++) {
        char c = infix[i];
        if (isalnum((unsigned char)c)) postfix[k++] = c;
        else if (c == '(') push(c);
        else if (c == ')') {
            while (top >= 0 && peek() != '(') postfix[k++] = pop();
            if (top >= 0 && peek() == '(') pop();
        } else {
            while (top >= 0 && peek() != '(' &&
                   (precedence(peek()) > precedence(c) ||
                   (precedence(peek()) == precedence(c) && c != '^')))
                postfix[k++] = pop();
            push(c);
        }
    }
    while (top >= 0) {
        char c = pop();
        if (c != '(') postfix[k++] = c;
    }
    postfix[k] = '\0';
    printf("Postfix expression: %s\n", postfix);
    return 0;
}
