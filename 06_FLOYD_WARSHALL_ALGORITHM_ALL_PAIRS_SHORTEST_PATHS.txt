FLOYD-WARSHALL ALGORITHM (ALL-PAIRS SHORTEST PATHS)
===================================================

#include <stdio.h>

#define MAX 100
#define INF 999999

int main(void) {
    int n, dist[MAX][MAX];
    printf("Enter number of vertices: ");
    scanf("%d", &n);
    if (n < 1 || n > MAX) return 1;

    printf("Enter adjacency matrix (%d for no edge):\n", INF);
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            scanf("%d", &dist[i][j]);

    for (int k = 0; k < n; k++)
        for (int i = 0; i < n; i++)
            for (int j = 0; j < n; j++)
                if (dist[i][k] != INF && dist[k][j] != INF &&
                    dist[i][j] > dist[i][k] + dist[k][j])
                    dist[i][j] = dist[i][k] + dist[k][j];

    printf("All-Pairs Shortest Path Matrix:\n");
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++)
            if (dist[i][j] == INF) printf("INF\t");
            else printf("%d\t", dist[i][j]);
        printf("\n");
    }
    return 0;
}
