HASH TABLE USING LINEAR PROBING
===============================

#include <stdio.h>

#define SIZE 10
int ht[SIZE], isEmpty[SIZE], isDeleted[SIZE];

int hash(int key) { return ((key % SIZE) + SIZE) % SIZE; }

void init(void) {
    for (int i = 0; i < SIZE; i++) {
        isEmpty[i] = 1;
        isDeleted[i] = 0;
        ht[i] = 0;
    }
}

void insert(int key) {
    int start = hash(key);
    for (int i = 0; i < SIZE; i++) {
        int idx = (start + i) % SIZE;
        if (isEmpty[idx] || isDeleted[idx]) {
            ht[idx] = key;
            isEmpty[idx] = 0;
            isDeleted[idx] = 0;
            printf("Inserted %d at index %d\n", key, idx);
            return;
        }
    }
    printf("Hash table is full\n");
}

void search(int key) {
    int start = hash(key);
    for (int i = 0; i < SIZE; i++) {
        int idx = (start + i) % SIZE;
        if (isEmpty[idx]) break;
        if (!isDeleted[idx] && ht[idx] == key) {
            printf("Element %d found at index %d\n", key, idx);
            return;
        }
    }
    printf("Element %d not found\n", key);
}

void deleteKey(int key) {
    int start = hash(key);
    for (int i = 0; i < SIZE; i++) {
        int idx = (start + i) % SIZE;
        if (isEmpty[idx]) break;
        if (!isDeleted[idx] && ht[idx] == key) {
            isDeleted[idx] = 1;
            printf("Deleted %d from index %d\n", key, idx);
            return;
        }
    }
    printf("Element %d not found\n", key);
}

void display(void) {
    printf("Index: ");
    for (int i = 0; i < SIZE; i++) printf("%d ", i);
    printf("\nValue: ");
    for (int i = 0; i < SIZE; i++) {
        if (isEmpty[i]) printf("_ ");
        else if (isDeleted[i]) printf("D ");
        else printf("%d ", ht[i]);
    }
    printf("\n");
}

int main(void) {
    int ch, key;
    init();
    while (1) {
        printf("\n1. Insert\n2. Search\n3. Delete\n4. Display\n5. Exit\nChoice: ");
        scanf("%d", &ch);
        switch (ch) {
            case 1: printf("Enter key: "); scanf("%d", &key); insert(key); break;
            case 2: printf("Enter key: "); scanf("%d", &key); search(key); break;
            case 3: printf("Enter key: "); scanf("%d", &key); deleteKey(key); break;
            case 4: display(); break;
            case 5: return 0;
            default: printf("Invalid choice\n");
        }
    }
}
