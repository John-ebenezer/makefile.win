HASH TABLE USING SEPARATE CHAINING
==================================

#include <stdio.h>
#include <stdlib.h>

#define SIZE 10
struct Node { int key; struct Node *next; };
struct Node *ht[SIZE] = {NULL};

int hash(int key) { return ((key % SIZE) + SIZE) % SIZE; }

void insert(int key) {
    int idx = hash(key);
    struct Node *node = malloc(sizeof(struct Node));
    if (!node) return;
    node->key = key;
    node->next = ht[idx];
    ht[idx] = node;
    printf("Inserted %d at index %d\n", key, idx);
}

void search(int key) {
    int idx = hash(key);
    struct Node *temp = ht[idx];
    while (temp) {
        if (temp->key == key) {
            printf("Element %d found at index %d\n", key, idx);
            return;
        }
        temp = temp->next;
    }
    printf("Element %d not found\n", key);
}

void deleteKey(int key) {
    int idx = hash(key);
    struct Node *temp = ht[idx], *prev = NULL;
    while (temp) {
        if (temp->key == key) {
            if (prev == NULL) ht[idx] = temp->next;
            else prev->next = temp->next;
            free(temp);
            printf("Deleted %d from index %d\n", key, idx);
            return;
        }
        prev = temp;
        temp = temp->next;
    }
    printf("Element %d not found\n", key);
}

void display(void) {
    for (int i = 0; i < SIZE; i++) {
        printf("%d: ", i);
        struct Node *temp = ht[i];
        while (temp) { printf("%d -> ", temp->key); temp = temp->next; }
        printf("NULL\n");
    }
}

int main(void) {
    int ch, key;
    while (1) {
        printf("\n1. Insert\n2. Search\n3. Delete\n4. Display\n5. Exit\nChoice: ");
        scanf("%d", &ch);
        switch (ch) {
            case 1: printf("Enter key: "); scanf("%d", &key); insert(key); break;
            case 2: printf("Enter key: "); scanf("%d", &key); search(key); break;
            case 3: printf("Enter key: "); scanf("%d", &key); deleteKey(key); break;
            case 4: display(); break;
            case 5: return 0;
            default: printf("Invalid choice\n");
        }
    }
}
