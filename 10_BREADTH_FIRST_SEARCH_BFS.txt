BREADTH FIRST SEARCH (BFS)
==========================

#include <stdio.h>

#define MAX 20
int adj[MAX][MAX], visited[MAX], queue[MAX];

int main(void) {
    int n, e, u, v, start, front = 0, rear = 0;
    printf("Enter number of vertices: ");
    scanf("%d", &n);
    if (n < 1 || n > MAX) return 1;
    printf("Enter number of edges: ");
    scanf("%d", &e);

    for (int i = 0; i < e; i++) {
        printf("Enter edge (u v): ");
        scanf("%d%d", &u, &v);
        if (u < 0 || u >= n || v < 0 || v >= n) { i--; continue; }
        adj[u][v] = adj[v][u] = 1;
    }

    printf("Enter starting vertex for BFS: ");
    scanf("%d", &start);
    if (start < 0 || start >= n) return 1;

    visited[start] = 1;
    queue[rear++] = start;
    printf("BFS Traversal: ");
    while (front < rear) {
        u = queue[front++];
        printf("%d ", u);
        for (v = 0; v < n; v++) {
            if (adj[u][v] && !visited[v]) {
                visited[v] = 1;
                queue[rear++] = v;
            }
        }
    }
    printf("\n");
    return 0;
}
