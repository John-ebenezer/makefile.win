DEPTH FIRST SEARCH (DFS) - RECURSIVE
====================================

#include <stdio.h>

#define MAX 20
int adj[MAX][MAX], visited[MAX], n;

void DFS(int u) {
    visited[u] = 1;
    printf("%d ", u);
    for (int v = 0; v < n; v++)
        if (adj[u][v] && !visited[v]) DFS(v);
}

int main(void) {
    int e, u, v, start;
    printf("Enter number of vertices: ");
    scanf("%d", &n);
    if (n < 1 || n > MAX) return 1;
    printf("Enter number of edges: ");
    scanf("%d", &e);

    for (int i = 0; i < e; i++) {
        printf("Enter edge (u v): ");
        scanf("%d%d", &u, &v);
        if (u < 0 || u >= n || v < 0 || v >= n) { i--; continue; }
        adj[u][v] = adj[v][u] = 1;
    }

    printf("Enter starting vertex for DFS: ");
    scanf("%d", &start);
    if (start < 0 || start >= n) return 1;
    printf("DFS Traversal: ");
    DFS(start);
    printf("\n");
    return 0;
}
