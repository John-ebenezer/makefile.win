DOUBLY LINKED LIST
==================

#include <stdio.h>
#include <stdlib.h>

struct Node { int data; struct Node *prev, *next; };

void insertBeginning(struct Node **head, int data) {
    struct Node *node = malloc(sizeof(struct Node));
    if (!node) return;
    node->data = data;
    node->prev = NULL;
    node->next = *head;
    if (*head) (*head)->prev = node;
    *head = node;
}

void insertEnd(struct Node **head, int data) {
    struct Node *node = malloc(sizeof(struct Node));
    if (!node) return;
    node->data = data;
    node->next = NULL;
    if (*head == NULL) {
        node->prev = NULL;
        *head = node;
        return;
    }
    struct Node *temp = *head;
    while (temp->next) temp = temp->next;
    temp->next = node;
    node->prev = temp;
}

void deleteNode(struct Node **head, int key) {
    struct Node *temp = *head;
    while (temp && temp->data != key) temp = temp->next;
    if (!temp) { printf("Element not found\n"); return; }
    if (temp->prev) temp->prev->next = temp->next;
    else *head = temp->next;
    if (temp->next) temp->next->prev = temp->prev;
    free(temp);
}

void display(struct Node *head) {
    if (!head) { printf("List is empty\n"); return; }
    printf("Doubly Linked List: ");
    while (head) {
        printf("%d", head->data);
        if (head->next) printf(" <-> ");
        head = head->next;
    }
    printf("\n");
}

int main(void) {
    struct Node *head = NULL;
    int ch, x;
    while (1) {
        printf("\n1. Insert at beginning\n2. Insert at end\n3. Delete node\n4. Display\n5. Exit\nChoice: ");
        scanf("%d", &ch);
        switch (ch) {
            case 1: printf("Enter data: "); scanf("%d", &x); insertBeginning(&head, x); break;
            case 2: printf("Enter data: "); scanf("%d", &x); insertEnd(&head, x); break;
            case 3: printf("Enter data to delete: "); scanf("%d", &x); deleteNode(&head, x); break;
            case 4: display(head); break;
            case 5: return 0;
            default: printf("Invalid choice\n");
        }
    }
}
