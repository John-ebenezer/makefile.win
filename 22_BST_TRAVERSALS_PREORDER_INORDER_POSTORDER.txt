BST TRAVERSALS (PREORDER, INORDER, POSTORDER)
=============================================

#include <stdio.h>
#include <stdlib.h>

struct Node { int data; struct Node *left, *right; };

struct Node *createNode(int x) {
    struct Node *node = malloc(sizeof(struct Node));
    if (!node) exit(EXIT_FAILURE);
    node->data = x;
    node->left = node->right = NULL;
    return node;
}

struct Node *insert(struct Node *root, int x) {
    if (!root) return createNode(x);
    if (x < root->data) root->left = insert(root->left, x);
    else if (x > root->data) root->right = insert(root->right, x);
    return root;
}

void preorder(struct Node *root) {
    if (root) { printf("%d ", root->data); preorder(root->left); preorder(root->right); }
}
void inorder(struct Node *root) {
    if (root) { inorder(root->left); printf("%d ", root->data); inorder(root->right); }
}
void postorder(struct Node *root) {
    if (root) { postorder(root->left); postorder(root->right); printf("%d ", root->data); }
}

int main(void) {
    struct Node *root = NULL;
    int n, x;
    printf("Enter number of nodes: ");
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        printf("Enter element: ");
        scanf("%d", &x);
        root = insert(root, x);
    }
    printf("Preorder: "); preorder(root);
    printf("\nInorder: "); inorder(root);
    printf("\nPostorder: "); postorder(root);
    printf("\n");
    return 0;
}
