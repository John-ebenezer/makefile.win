HEAP OPERATIONS (MAX HEAP: CREATION, INSERTION, DELETE MAX)
===========================================================

#include <stdio.h>

#define MAX 100
int heap[MAX], size = 0;

void heapifyDown(int i) {
    while (1) {
        int largest = i, left = 2 * i + 1, right = 2 * i + 2;
        if (left < size && heap[left] > heap[largest]) largest = left;
        if (right < size && heap[right] > heap[largest]) largest = right;
        if (largest == i) break;
        int temp = heap[i]; heap[i] = heap[largest]; heap[largest] = temp;
        i = largest;
    }
}

void buildHeap(int arr[], int n) {
    size = n;
    for (int i = 0; i < n; i++) heap[i] = arr[i];
    for (int i = size / 2 - 1; i >= 0; i--) heapifyDown(i);
}

void insertHeap(int x) {
    if (size == MAX) { printf("Heap is full\n"); return; }
    int i = size++;
    heap[i] = x;
    while (i > 0 && heap[(i - 1) / 2] < heap[i]) {
        int p = (i - 1) / 2;
        int temp = heap[i]; heap[i] = heap[p]; heap[p] = temp;
        i = p;
    }
}

int deleteHeap(void) {
    if (size == 0) { printf("Heap is empty\n"); return -1; }
    int max = heap[0];
    heap[0] = heap[--size];
    if (size > 0) heapifyDown(0);
    return max;
}

void display(void) {
    if (!size) { printf("Heap is empty\n"); return; }
    printf("Heap: ");
    for (int i = 0; i < size; i++) printf("%d ", heap[i]);
    printf("\n");
}

int main(void) {
    int n, arr[MAX], ch, x;
    printf("Enter number of elements (0-%d): ", MAX);
    scanf("%d", &n);
    if (n < 0 || n > MAX) return 1;
    printf("Enter elements: ");
    for (int i = 0; i < n; i++) scanf("%d", &arr[i]);
    buildHeap(arr, n);

    while (1) {
        printf("\n1. Insert\n2. Delete (Extract Max)\n3. Display\n4. Exit\nChoice: ");
        scanf("%d", &ch);
        switch (ch) {
            case 1: printf("Enter element: "); scanf("%d", &x); insertHeap(x); break;
            case 2: x = deleteHeap(); if (x != -1) printf("Deleted max: %d\n", x); break;
            case 3: display(); break;
            case 4: return 0;
            default: printf("Invalid choice\n");
        }
    }
}
